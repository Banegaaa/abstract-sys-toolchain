// get the following token (section name)
i++;
// check if it exists in the symbol table
// if it does, set the is_section to true, set the section to the current section and set the lc_value to the current LC
bool exists = false;
for (int j = 0; j < symbolTable.size(); j++){
    if (symbolTable[j].name == tokenList[i].name){
        exists = true;
        symbolTable[j].is_section = true;
        symbolTable[j].section = currentSection;
        symbolTable[j].lc_value = LC;
    }
}
if (!exists){
    // if it does not, add it to the symbol table with the section set to the current section, the lc_value set to the current LC, the is_global set to false and the is_section set to true
    symbolTable.push_back(Symbol(tokenList[i].name, currentSection, LC, false, true, symbolNum, true));
    symbolNum++;
    
    // increment the current section and reset the LC to 0
    currentSection++;
    LC = 0;
    // 
}
